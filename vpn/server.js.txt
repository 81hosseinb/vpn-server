const express = require('express'); const cors = require('cors'); const app = express(); let vpns = []; let idCounter = 1; app.use(cors()); app.use(express.json()); // میدلور احراز هویت ادمین const ADMIN_TOKEN = process.env.ADMIN_TOKEN || 'your-secret-token'; const adminAuth = (req, res, next) => { const token = req.headers['x-admin-token']; if (token !== ADMIN_TOKEN) return res.status(401).json({error: 'Unauthorized'}); next(); }; // دریافت همه VPN های فعال app.get('/api/vpns', (req, res) => { const active = vpns.filter(v => v.active) .sort((a,b) => b.priority - a.priority); res.json(active); }); // یک VPN تصادفی فعال app.get('/api/vpns/random', (req, res) => { const active = vpns.filter(v => v.active); if (!active.length) return res.status(404).json({error: 'No VPN available'}); const vpn = active[Math.floor(Math.random() * active.length)]; res.json(vpn); }); // افزودن VPN (فقط ادمین) app.post('/api/vpns', adminAuth, (req, res) => { const {name, link, type, country, priority} = req.body; if (!name || !link) return res.status(400).json({error: 'name and link required'}); const newVpn = {id: idCounter++, name, link, type, country, priority: priority||1, active: true}; vpns.push(newVpn); res.status(201).json(newVpn); }); // حذف VPN (فقط ادمین) app.delete('/api/vpns/:id', adminAuth, (req, res) => { vpns = vpns.filter(v => v.id !== +req.params.id); res.json({success: true}); }); app.listen(3000, () => console.log('Server running on port 3000'));
